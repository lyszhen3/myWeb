# 基础配置
~~~~
" 去除提示音
set vb t_vb=
au GuiEnter * set t_vb=

set guifont=courier_new:h11
set nu
" 忽略大小写
set ic
set enc=UTF-8
set langmenu=zh_CN.UTF-8
source $VIMRUNTIME/delmenu.vim
source $VIMRUNTIME/menu.vim

set noundofile
set nobackup
set noswapfile
set tabstop=4
set shiftwidth=4

vnoremap <C-C> "+y
vnoremap <C-X> "+x
noremap <C-Q> <C-V>
noremap <C-V> "+gp
cmap <C-V> <C-R>+

set fileencodings=ucs-bom,utf-8,cp936,gb18030,big5,euc-jp,euc-kr,latin1
set encoding=utf-8
set termencoding=cp936
language messages zh_CN.UTF-8

" 使用 :Jsonf json格式化
command! Jsonf :execute '%!python -m json.tool --no-ensure-ascii'
~~~~

# 基础操作

* json 序列化
  * 通过scoop 安装 jq
    * %!jq .
  * 或_vimrc配置后使用  :Jsonf 格式化
  * 折叠
    * :set foldmethod=syntax 根据高亮折叠
    * zm 折叠全部
    * zr 展开全部
    * zc 折叠当前行
    * zo 展开当前行

* 16 进制
  * %!xxd
    * %!xxd -r 退出
  * 或 Hexmode (需要fidian/hexmode 插件)
    *:Hexmode 进入 :Hexmode 退出  
    * Plugin 'fidian/hexmode'

* markdown预览
  * Plugin 'iamcco/markdown-preview.nvim', { 'do': 'cd app && npm install' }
  * :PluginInstall
  * :call mkdp#util#install()
  * :MarkdownPreview启动预览，:MarkdownPreviewStop 停止预览
